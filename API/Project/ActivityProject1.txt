package github;

import io.restassured.builder.RequestSpecBuilder;
import io.restassured.http.ContentType;
import io.restassured.response.Response;
import io.restassured.specification.RequestSpecification;

import org.testng.Assert;
import org.testng.Reporter;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import static io.restassured.RestAssured.given;

public class GitHubSSHKeyTest {

    private RequestSpecification requestSpecification;

    // SSH public key
    private String sshKey =
            "ssh-ed25519 AAAAC3NzaC1lZDI1NTE5AAAAIYOUR_PUBLIC_KEY";

    // ID generated by GitHub
    private int keyId;


    @BeforeClass
    public void setup() {

        String githubToken = System.getenv("GITHUB_TOKEN");

        if (githubToken == null || githubToken.isBlank()) {
            throw new RuntimeException(
                    "GITHUB_TOKEN environment variable is not set."
            );
        }

        requestSpecification = new RequestSpecBuilder()
                .setBaseUri("https://api.github.com")
                .setContentType(ContentType.JSON)
                .addHeader(
                        "Authorization",
                        "Bearer " + githubToken
                )
                .build();
    }


    // ============================================================
    // POST - Create SSH Key
    // ============================================================

    @Test(priority = 1)
    public void createSSHKey() {

        String requestBody =
                "{"
                + "\"title\":\"TestAPIKey\","
                + "\"key\":\"" + sshKey + "\""
                + "}";


        Response response =
                given()
                        .spec(requestSpecification)
                        .body(requestBody)

                .when()
                        .post("/user/keys")

                .then()
                        .extract()
                        .response();


        // Print response
        Reporter.log(
                "POST Response: " + response.asPrettyString(),
                true
        );


        // Status code assertion
        Assert.assertEquals(
                response.getStatusCode(),
                201,
                "SSH key creation failed"
        );


        // Extract generated key ID
        keyId = response
                .jsonPath()
                .getInt("id");


        Reporter.log(
                "Generated SSH Key ID: " + keyId,
                true
        );


        // Verify ID was generated
        Assert.assertTrue(
                keyId > 0,
                "SSH Key ID was not generated"
        );
    }


    // ============================================================
    // GET - Get SSH Key
    // ============================================================

    @Test(priority = 2, dependsOnMethods = "createSSHKey")
    public void getSSHKey() {

        Response response =
                given()
                        .spec(requestSpecification)
                        .pathParam("keyId", keyId)

                .when()
                        .get("/user/keys/{keyId}")

                .then()
                        .extract()
                        .response();


        // Print response
        Reporter.log(
                "GET Response: " + response.asPrettyString(),
                true
        );


        // Status code assertion
        Assert.assertEquals(
                response.getStatusCode(),
                200,
                "GET SSH key request failed"
        );


        // Verify returned ID
        int returnedKeyId =
                response.jsonPath().getInt("id");

        Assert.assertEquals(
                returnedKeyId,
                keyId,
                "Returned key ID does not match created key ID"
        );


        // Verify title
        String title =
                response.jsonPath().getString("title");

        Assert.assertEquals(
                title,
                "TestAPIKey",
                "SSH key title does not match"
        );
    }


    // ============================================================
    // DELETE - Delete SSH Key
    // ============================================================

    @Test(priority = 3, dependsOnMethods = "getSSHKey")
    public void deleteSSHKey() {

        Response response =
                given()
                        .spec(requestSpecification)
                        .pathParam("keyId", keyId)

                .when()
                        .delete("/user/keys/{keyId}")

                .then()
                        .extract()
                        .response();


        // Print response
        Reporter.log(
                "DELETE Response Status: "
                        + response.getStatusCode(),
                true
        );


        // Status code assertion
        Assert.assertEquals(
                response.getStatusCode(),
                204,
                "SSH key deletion failed"
        );
    }
}
